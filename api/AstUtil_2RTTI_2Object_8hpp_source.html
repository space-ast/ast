395// 特化：仅当 &T::getType 的类型严格为 Class* (T::*)()const 时匹配

400template<typename T>
401T aobject_cast(Object* obj)
402{
403 using ObjectType = typename std::decay<typename std::remove_pointer<T>::type>::type;
404 static_assert(has_own_getType<ObjectType>::value, "aobject_cast requires the type to has a AST_OBJECT macro");
405 return static_cast<T>(ObjectType::StaticType()->cast(obj));
406}
