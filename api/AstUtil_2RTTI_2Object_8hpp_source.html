425// 特化：仅当 &T::getType 的类型严格为 Class* (T::*)()const 时匹配
426template <typename T>
427struct has_own_getType<T, typename std::enable_if<
428 std::is_same<decltype(&T::getType), Class* (T::*)()const>::value>::type> : std::true_type {};

430template<typename T>
431T aobject_cast(Object* obj)
432{
433 using ObjectType = typename std::decay<typename std::remove_pointer<T>::type>::type;
434 static_assert(has_own_getType<ObjectType>::value, "aobject_cast requires the type to has a AST_OBJECT macro");
435 return static_cast<T>(ObjectType::StaticType()->cast(obj));
436}

438template<typename T>
439bool Object::isOfType() const
440{
441 using ObjectType = typename std::decay<typename std::remove_pointer<T>::type>::type;
442 static_assert(has_own_getType<ObjectType>::value, "isOfType requires the type to has a AST_OBJECT macro");
443 return isOfType(ObjectType::StaticType());
444}
