426// 定义：检测 T 是否拥有 “Class* (T::*)() const” 签名的成员函数 getType
427template <typename T>
428struct has_own_getType {
429private:
430 typedef char yes;
431 typedef long no;
432
433 // 仅当 U 具有 Class* (U::*)() const 的成员时，此辅助模板才能实例化
434 template <typename U, Class* (U::*)() const>
435 struct Check;
436
437 // 如果能匹配 &U::getType，则返回 yes
438 template <typename U>
439 static yes test(Check<U, &U::getType>*);
440
441 // 否则走 ...
442 template <typename>
443 static no test(...);
444
445public:
446 static constexpr bool value = (sizeof(test<T>(0)) == sizeof(yes));
447};

450template<typename T, typename U>
451A_ALWAYS_INLINE T aobject_cast(U* obj)
452{
453 // obj 可以是 Object* 或 const Object*
454 using ObjectType = typename std::decay<
455 typename std::remove_pointer<
456 typename std::remove_pointer<T>::type
457 >::type
458 >::type;
459 static_assert(has_own_getType<ObjectType>::value,
460 "aobject_cast requires the type to has a AST_OBJECT macro");
461 return static_cast<T>(ObjectType::StaticType()->cast(obj));
462}
