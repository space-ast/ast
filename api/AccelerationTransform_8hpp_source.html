198A_ALWAYS_INLINE AccelerationTransform AccelerationTransform::composed(const AccelerationTransform &next) const
199{
205 // 先取到局部量，再构造返回值，避免原地赋值
206 Matrix3d mat1 = this->matrix();
207 Vector3d angvel1 = this->rotationRate();
208 Vector3d angvelDot1 = this->rotationRateDot();
209 Vector3d tau = next.translation() * mat1; // 即 M₁ᵀ t₂
210 Vector3d velocity2 = next.velocity() * mat1; // 即 M₁ᵀ V₂
211 Vector3d acceleration2 = next.acceleration() * mat1; // 即 M₁ᵀ A₂
212
213 Vector3d translation = this->translation() + tau;
214 Vector3d velocity = this->velocity() + velocity2 + angvel1.cross(tau);
215 Vector3d acceleration = this->acceleration() + acceleration2
216 + angvel1.cross(velocity2) * 2.0
217 + angvel1.cross(angvel1.cross(tau))
218 + angvelDot1.cross(tau);
219
220 AccelerationRotation rotation = AccelerationRotation(mat1, angvel1, angvelDot1)
221 .composed(next.accelerationRotation());
222
223 return AccelerationTransform(translation, velocity, acceleration, rotation);
224}

242A_ALWAYS_INLINE void AccelerationTransform::getInverse(AccelerationTransform &inversed) const
243{
248 // 先全部取到局部量，允许 inversed 与 *this 为同一对象
249 AccelerationRotation rotation = this->accelerationRotation();
250 AccelerationRotation inversedRotation;
251 rotation.getInverse(inversedRotation);
252
253 rotation.transformVecVelAcc(
254 -this->translation(),
255 -this->velocity(),
256 -this->acceleration(),
257 inversed.translation(),
258 inversed.velocity(),
259 inversed.acceleration_);
260
261 inversed.setAccelerationRotation(inversedRotation);
262}
