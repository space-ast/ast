237A_ALWAYS_INLINE AccelerationTransform AccelerationTransform::composed(const AccelerationTransform &next) const
238{
244 // 先取到局部量，再构造返回值，避免原地赋值
245 Matrix3d mat1 = this->getMatrix();
246 Vector3d angvel1 = this->getRotationRate();
247 Vector3d angvelDot1 = this->rotationRateDot();
248 Vector3d tau = next.getTranslation() * mat1; // 即 M₁ᵀ t₂
249 Vector3d velocity2 = next.getVelocity() * mat1; // 即 M₁ᵀ V₂
250 Vector3d acceleration2 = next.getAcceleration() * mat1; // 即 M₁ᵀ A₂
251
252 Vector3d translation = this->getTranslation() + tau;
253 Vector3d velocity = this->getVelocity() + velocity2 + angvel1.cross(tau);
254 Vector3d acceleration = this->getAcceleration() + acceleration2
255 + angvel1.cross(velocity2) * 2.0
256 + angvel1.cross(angvel1.cross(tau))
257 + angvelDot1.cross(tau);
258
259 AccelerationRotation rotation = AccelerationRotation(mat1, angvel1, angvelDot1)
260 .composed(next.getAccelerationRotation());
261
262 return AccelerationTransform(translation, velocity, acceleration, rotation);
263}

281A_ALWAYS_INLINE void AccelerationTransform::getInverse(AccelerationTransform &inversed) const
282{
287 // 先全部取到局部量，允许 inversed 与 *this 为同一对象
288 AccelerationRotation rotation = this->getAccelerationRotation();
289 AccelerationRotation inversedRotation;
290 rotation.getInverse(inversedRotation);
291
292 rotation.transformVecVelAcc(
293 -this->getTranslation(),
294 -this->getVelocity(),
295 -this->getAcceleration(),
296 inversed.getTranslation(),
297 inversed.getVelocity(),
298 inversed.acceleration_);
299
300 inversed.setAccelerationRotation(inversedRotation);
301}
