138template<typename _Scalar>
139inline double VectorN<_Scalar, 3>::Angle(const Self& v1, const Self & v2)
140{
141 double normProduct = v1.norm() * v2.norm();
142 if (normProduct == 0) {
143 return 0;
144 }
145
146 double dot = v1.dot(v2);
147 double threshold = normProduct * 0.9999;
148 if ((dot < -threshold) || (dot > threshold)) {
149 Self v3 = v1.cross(v2);
150 if (dot >= 0) {
151 return asin(v3.norm() / normProduct);
152 }
153 return kPI - asin(v3.norm() / normProduct);
154 }
155
156 return acos(dot / normProduct);
157}
