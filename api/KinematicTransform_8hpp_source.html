155A_ALWAYS_INLINE KinematicTransform KinematicTransform::composed(const KinematicTransform &next) const
156{
157 KinematicRotation rotation = this->kinematicRotation().composed(next.kinematicRotation());
158 /*
159 也可以通过 getRotation().transformVectorVelocityInv() 来实现。
160 */
161 Vector3d vector = next.translation() * this->matrix();
162
163 Vector3d translation = this->translation() + vector;
164 Vector3d velocity = this->velocity() + next.velocity() * this->matrix()
165 + this->rotationRate().cross(vector);
166 return KinematicTransform(translation, velocity, rotation);
167}
