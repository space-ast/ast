160A_ALWAYS_INLINE KinematicTransform KinematicTransform::composed(const KinematicTransform &next) const
161{
162 KinematicRotation rotation = this->getKinematicRotation().composed(next.getKinematicRotation());
163 /*
164 也可以通过 getRotation().transformVectorVelocityInv() 来实现。
165 */
166 Vector3d vector = next.getTranslation() * this->getMatrix();
167
168 Vector3d translation = this->getTranslation() + vector;
169 Vector3d velocity = this->getVelocity() + next.getVelocity() * this->getMatrix()
170 + this->getRotationRate().cross(vector);
171 return KinematicTransform(translation, velocity, rotation);
172}
