179A_ALWAYS_INLINE AccelerationRotation AccelerationRotation::composed(const AccelerationRotation &next) const
180{
187 Matrix3d mat1 = this->matrix_;
188 Vector3d angvel1 = this->angvel_;
189 Vector3d angvel2InSource = next.angvel_ * mat1; // 即 M1ᵀ ω2
190 Vector3d angvel = angvel1 + angvel2InSource;
191 Vector3d angvelDot = this->angvelDot_ + angvel1.cross(angvel2InSource) + next.angvelDot_ * mat1;
192 return AccelerationRotation(next.matrix_ * mat1, angvel, angvelDot);
193}

211A_ALWAYS_INLINE void AccelerationRotation::getInverse(AccelerationRotation &inversed) const
212{
213 // 先取出matrix_，允许inversed与*this为同一对象
214 Matrix3d mat = this->matrix_;
215 Vector3d angvel = this->angvel_;
216 Vector3d angvelDot = this->angvelDot_;
217 inversed.matrix_ = mat.transpose();
218 inversed.angvel_ = -(mat * angvel);
219 inversed.angvelDot_ = -(mat * angvelDot); // ω×ω = 0，故角加速度没有附加项
220}

229A_ALWAYS_INLINE void AccelerationRotation::transformVecVelAcc(
230 const Vector3d &position, const Vector3d &velocity, const Vector3d &acceleration,
231 Vector3d &positionOut, Vector3d &velocityOut, Vector3d &accelerationOut) const
232{
233 // 注意：这里要先计算accelerationOut，再计算velocityOut，最后计算positionOut，
234 // 防止入参与出参地址相同时值被覆盖
235 accelerationOut = this->matrix_ * (acceleration
236 - this->angvel_.cross(velocity) * 2.0
237 - this->angvelDot_.cross(position)
238 + this->angvel_.cross(this->angvel_.cross(position)));
239 velocityOut = this->matrix_ * (velocity - this->angvel_.cross(position));
240 positionOut = this->matrix_ * position;
241}

243A_ALWAYS_INLINE void AccelerationRotation::transformVecVelAccInv(
244 const Vector3d &position, const Vector3d &velocity, const Vector3d &acceleration,
245 Vector3d &positionOut, Vector3d &velocityOut, Vector3d &accelerationOut) const
246{
247 // 注意：这里要先计算positionOut，再计算velocityOut，最后计算accelerationOut
248 positionOut = position * this->matrix_;
249 velocityOut = velocity * this->matrix_ + this->angvel_.cross(positionOut);
250 accelerationOut = acceleration * this->matrix_
251 + this->angvelDot_.cross(positionOut)
252 + this->angvel_.cross(velocityOut) * 2.0
253 - this->angvel_.cross(this->angvel_.cross(positionOut));
254}
