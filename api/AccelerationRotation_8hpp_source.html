177A_ALWAYS_INLINE AccelerationRotation AccelerationRotation::composed(const AccelerationRotation &next) const
178{
185 Matrix3d mat1 = this->matrix();
186 Vector3d angvel1 = this->angvel_;
187 Vector3d angvel2InSource = next.angvel_ * mat1; // 即 M1ᵀ ω2
188 Vector3d angvel = angvel1 + angvel2InSource;
189 Vector3d angvelDot = this->angvelDot_ + angvel1.cross(angvel2InSource) + next.angvelDot_ * mat1;
190 return AccelerationRotation(next.matrix() * mat1, angvel, angvelDot);
191}

209A_ALWAYS_INLINE void AccelerationRotation::getInverse(AccelerationRotation &inversed) const
210{
211 // 先取出matrix_，允许inversed与*this为同一对象
212 Matrix3d mat = this->matrix();
213 Vector3d angvel = this->angvel_;
214 Vector3d angvelDot = this->angvelDot_;
215 inversed.matrix() = mat.transpose();
216 inversed.angvel_ = -(mat * angvel);
217 inversed.angvelDot_ = -(mat * angvelDot); // ω×ω = 0，故角加速度没有附加项
218}

227A_ALWAYS_INLINE void AccelerationRotation::transformVecVelAcc(
228 const Vector3d &position, const Vector3d &velocity, const Vector3d &acceleration,
229 Vector3d &positionOut, Vector3d &velocityOut, Vector3d &accelerationOut) const
230{
231 // 注意：这里要先计算accelerationOut，再计算velocityOut，最后计算positionOut，
232 // 防止入参与出参地址相同时值被覆盖
233 accelerationOut = this->matrix() * (acceleration
234 - this->angvel_.cross(velocity) * 2.0
235 - this->angvelDot_.cross(position)
236 + this->angvel_.cross(this->angvel_.cross(position)));
237 velocityOut = this->matrix() * (velocity - this->angvel_.cross(position));
238 positionOut = this->matrix() * position;
239}

241A_ALWAYS_INLINE void AccelerationRotation::transformVecVelAccInv(
242 const Vector3d &position, const Vector3d &velocity, const Vector3d &acceleration,
243 Vector3d &positionOut, Vector3d &velocityOut, Vector3d &accelerationOut) const
244{
245 // 注意：这里要先计算positionOut，再计算velocityOut，最后计算accelerationOut
246 positionOut = position * this->matrix();
247 velocityOut = velocity * this->matrix() + this->angvel_.cross(positionOut);
248 accelerationOut = acceleration * this->matrix()
249 + this->angvelDot_.cross(positionOut)
250 + this->angvel_.cross(velocityOut) * 2.0
251 - this->angvel_.cross(this->angvel_.cross(positionOut));
252}
