297inline TimeInterval::DiscreteTimePointRange TimeInterval::discrete(double step) const
298{
299 double dur = duration();
300 if (step <= 0.0 || dur <= 0.0) {
301 return DiscreteTimePointRange(*this, step, 0);
302 }
303 size_t n = static_cast<size_t>(std::ceil(dur / step)) + 1;
304 return DiscreteTimePointRange(*this, step, n);
305}

307inline TimeInterval::DiscreteEpochSecondRange TimeInterval::discrete(const TimePoint& epoch, double step) const
308{
309 double dur = duration();
310 if (step <= 0.0 || dur <= 0.0) {
311 return DiscreteEpochSecondRange(0.0, step, 0.0, 0);
312 }
313 size_t n = static_cast<size_t>(std::ceil(dur / step)) + 1;
314 double offset = getStart() - epoch;
315 double stopOffset = getStop() - epoch;
316 return DiscreteEpochSecondRange(offset, step, stopOffset, n);
317}

319inline errc_t TimeInterval::merge(const TimeInterval &other)
320{
321 const TimePoint& thisStart = start();
322 const TimePoint& otherStart = other.start();
323 TimePoint thisStop = stop();
324 TimePoint otherStop = other.stop();
325
326 if (thisStart.durationFrom(otherStop) > 0 || otherStart.durationFrom(thisStop) > 0)
327 {
328 aError("merge time interval failed, no overlap");
329 return eErrorInvalidParam;
330 }
331
332 const TimePoint& mergedStart = (thisStart.durationFrom(otherStart) <= 0) ? thisStart : otherStart;
333 const TimePoint& mergedStop = (thisStop.durationFrom(otherStop) >= 0) ? thisStop : otherStop;
334
335 setStartStop(mergedStart, mergedStop);
336 return eNoError;
337}
