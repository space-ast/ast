354inline TimeInterval::DiscreteTimePointRange TimeInterval::discrete(double step) const
355{
356 double dur = duration();
357 if (step <= 0.0 || dur <= 0.0) {
358 return DiscreteTimePointRange(*this, step, 0);
359 }
360 size_t n = static_cast<size_t>(std::ceil(dur / step)) + 1;
361 return DiscreteTimePointRange(*this, step, n);
362}

364inline TimeInterval::DiscreteEpochSecondRange TimeInterval::discrete(const TimePoint& epoch, double step) const
365{
366 double dur = duration();
367 if (step <= 0.0 || dur <= 0.0) {
368 return DiscreteEpochSecondRange(0.0, step, 0.0, 0);
369 }
370 size_t n = static_cast<size_t>(std::ceil(dur / step)) + 1;
371 double offset = getStart() - epoch;
372 double stopOffset = getStop() - epoch;
373 return DiscreteEpochSecondRange(offset, step, stopOffset, n);
374}

401inline TimeInterval& TimeInterval::intersect(const TimeInterval &other)
402{
403 const TimePoint& thisStart = this->start();
404 const TimePoint& thisStop = this->stop();
405 const TimePoint& otherStart = other.start();
406 const TimePoint& otherStop = other.stop();
407

413 {
414 this->setStop(otherStop);
415 }
416

418 if(this->start() > this->stop())

432inline bool TimeInterval::intersects(const TimeInterval &other) const
433{
434 // 零长度区间（start == stop）语义为空，不应与任何区间相交
435 auto& start = (std::max)(start_, other.start());
436 auto& stop = (std::min)(stop_, other.stop());
437 return start < stop;
438}
