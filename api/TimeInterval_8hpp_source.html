302inline TimePointRange TimeInterval::discretize(double step) const
303{
304 double dur = duration();
305 if (step > 0 && dur >= 0)
306 {
307 size_t n = static_cast<size_t>(std::ceil(dur / step)) + 1;
308 return TimePointRange(start_, stop_, step, n);
309 }
310 return TimePointRange(start_, stop_, step, 0);
311}

313inline DoubleRange TimeInterval::discretize(const TimePoint& epoch, double step) const
314{
315 double dur = duration();
316 if (step > 0 && dur >= 0)
317 {
318 size_t n = static_cast<size_t>(std::ceil(dur / step)) + 1;
319 double start = getStart() - epoch;
320 double stop = getStop() - epoch;
321 return DoubleRange(start, stop, step, n);
322 }
323
324 return DoubleRange(0.0, 0.0, step, 0);
325}
