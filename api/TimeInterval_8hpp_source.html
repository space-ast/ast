294inline TimeInterval::DiscreteTimePointRange TimeInterval::discrete(double step) const
295{
296 double dur = duration();
297 if (step <= 0.0 || dur <= 0.0) {
298 return DiscreteTimePointRange(*this, step, 0);
299 }
300 size_t n = static_cast<size_t>(std::ceil(dur / step)) + 1;
301 return DiscreteTimePointRange(*this, step, n);
302}

304inline TimeInterval::DiscreteEpochSecondRange TimeInterval::discrete(const TimePoint& epoch, double step) const
305{
306 double dur = duration();
307 if (step <= 0.0 || dur <= 0.0) {
308 return DiscreteEpochSecondRange(0.0, step, 0.0, 0);
309 }
310 size_t n = static_cast<size_t>(std::ceil(dur / step)) + 1;
311 double offset = getStart() - epoch;
312 double stopOffset = getStop() - epoch;
313 return DiscreteEpochSecondRange(offset, step, stopOffset, n);
314}

316inline errc_t TimeInterval::merge(const TimeInterval &other)
317{
318 const TimePoint& thisStart = start();
319 const TimePoint& otherStart = other.start();
320 TimePoint thisStop = stop();
321 TimePoint otherStop = other.stop();
322
323 if (thisStart.durationFrom(otherStop) > 0 || otherStart.durationFrom(thisStop) > 0)
324 {
325 aError("merge time interval failed, no overlap");
326 return eErrorInvalidParam;
327 }
328
329 const TimePoint& mergedStart = (thisStart.durationFrom(otherStart) <= 0) ? thisStart : otherStart;
330 const TimePoint& mergedStop = (thisStop.durationFrom(otherStop) >= 0) ? thisStop : otherStop;
331
332 setStartStop(mergedStart, mergedStop);
333 return eNoError;
334}
