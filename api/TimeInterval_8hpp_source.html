269inline TimePointRange TimeInterval::discretize(double step) const
270{
271 double dur = duration();
272 if (step <= 0.0 || dur <= 0.0) {
273 return TimePointRange(start_, stop_, step, 0);
274 }
275 size_t n = static_cast<size_t>(std::ceil(dur / step)) + 1;
276 return TimePointRange(start_, stop_, step, n);
277}

279inline DoubleRange TimeInterval::discretize(const TimePoint& epoch, double step) const
280{
281 double dur = duration();
282 if (step <= 0.0 || dur <= 0.0) {
283 return DoubleRange(0.0, 0.0, step, 0);
284 }

286 double start = getStart() - epoch;
287 double stop = getStop() - epoch;
288 return DoubleRange(start, stop, step, n);

316inline TimeInterval& TimeInterval::intersect(const TimeInterval &other)
317{
318 const TimePoint& thisStart = this->start();
319 const TimePoint& thisStop = this->stop();
320 const TimePoint& otherStart = other.start();
321 const TimePoint& otherStop = other.stop();
322
323 if(otherStart > thisStart)
324 {
325 this->setStart(otherStart);
326 }
327 if(otherStop < thisStop)
328 {
329 this->setStop(otherStop);
330 }
331
332 // 空交集 → 零长度区间（避免负 duration）
333 if(this->start() > this->stop())
334 {
335 this->setStop(this->start());
336 }
337 return *this;
338}

347inline bool TimeInterval::intersects(const TimeInterval &other) const
348{
349 // 零长度区间（start == stop）语义为空，不应与任何区间相交
350 auto& start = (std::max)(start_, other.start());
351 auto& stop = (std::min)(stop_, other.stop());
352 return start < stop;
353}
