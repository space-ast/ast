171 template<typename T, typename... Args>
172 typename std::enable_if<!std::is_reference<T>::value, T&>::type
173 emplace(Args&&... args)
174 {
175 reset();
176 void* storage = ::operator new(sizeof(T));
177 try
178 {
179 ::new (storage) T(std::forward<Args>(args)...);
180 }
181 catch (...)
182 {
183 ::operator delete(storage);
184 throw;
185 }
186 ops_ = &opsFor<T>();
187 data_ = storage;
188 return *static_cast<T*>(data_);
189 }
190
192 void swap(Any& other) noexcept
193 {
194 std::swap(ops_, other.ops_);
195 std::swap(data_, other.data_);
196 }

199 // ---- 类型擦除操作表 ----
200 struct TypeOps
201 {
202 void (*destroy)(void*);
203 void (*copyConstruct)(void* dst, const void* src);
204 const std::type_info* type;
205 size_t size;
206 };
207
208 template<typename T>
209 static const TypeOps& opsFor()
210 {
211 static const TypeOps ops = {
212 [](void* p) { static_cast<T*>(p)->~T(); },
213 [](void* dst, const void* src) { ::new (dst) T(*static_cast<const T*>(src)); },
214 &typeid(T),
215 sizeof(T),
216 };
217 return ops;
218 }

220 void copyFrom(const Any& other)
221 {
222 if (!other.data_)
223 return;
224 void* storage = ::operator new(other.ops_->size);
225 try
226 {
227 other.ops_->copyConstruct(storage, other.data_);
228 }
229 catch (...)
230 {
231 ::operator delete(storage);
232 throw;
233 }
234 ops_ = other.ops_;
235 data_ = storage;
236 }
237
238 void stealFrom(Any& other) noexcept
239 {
240 ops_ = other.ops_;
241 data_ = other.data_;
242 other.ops_ = nullptr;
243 other.data_ = nullptr;
244 }
