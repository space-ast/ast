73class AST_UTIL_API CollectingStreamReceiver : public NetworkStreamReceiver
74{
75public:
76 explicit CollectingStreamReceiver(NetworkResponse& response)
77 : response_(response), body_() {}
78
79 void onHeaders(int statusCode,
80 const std::map<std::string, std::string>& headers) override
81 {
82 response_.setStatusCode(statusCode);
83 response_.setHeaders(headers);
84 }
85
86 errc_t onData(const char* data, size_t size) override
87 {
88 // 防止无限制响应体耗尽内存（上限 100 MB）
89 if (body_.size() + size > kMaxBodySize)
90 return eErrorOutOfRange;
91 body_.append(data, size);
92 return 0;
93 }
94
95 void onComplete() override
96 {
97 response_.setBody(body_);
98 }
99
100 void onError(errc_t /*error*/) override
101 {
102 // 保留已累积的部分体数据（用于诊断）
103 if (!body_.empty())
104 response_.setBody(body_);
105 }
106
107private:
108 static constexpr size_t kMaxBodySize = 100 * 1024 * 1024; // 100 MB
109 NetworkResponse& response_;
110 std::string body_;
111};
