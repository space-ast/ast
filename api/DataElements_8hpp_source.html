47 ElementExtractFunc makeExtractFunc()
48 {
49 return [](const VariantVector& data, VariantVector& element) -> errc_t
50 {
51 auto span = data.asSpan<const T>();
52 if (!span.data()) return -1;
53 using ElementType = std::remove_cv_t<std::remove_reference_t<MemberType>>;
54 element.reset<ElementType>(span.size());
55 auto* out = element.as<ElementType>();
56 if (!out) return -1;
57 for (size_t i = 0; i < span.size(); ++i)
58 out[i] = span[i].*Member;
59 return 0;
60 };
61 }
