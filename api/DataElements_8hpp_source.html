48 template<typename T, typename MemberType, MemberType T::*Member>
49 ElementExtractFunc makeExtractFunc()
50 {
51 return [](const VariantVector& data, VariantVector& element) -> errc_t
52 {
53 auto span = data.asSpan<const T>();
54 if (!span.data()) return -1;
55 using ElementType = typename std::remove_cv<typename std::remove_reference<MemberType>::type>::type;
56 element.reset<ElementType>(span.size());
57 auto* out = element.as<ElementType>();
58 if (!out) return -1;
59 for (size_t i = 0; i < span.size(); ++i)
60 out[i] = span[i].*Member;
61 return 0;
62 };
63 }
