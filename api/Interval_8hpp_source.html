162inline DoubleRange Interval::discretize(double step) const
163{
164 double dur = duration();
165 if (step <= 0.0 || dur <= 0.0) {
166 return DoubleRange(0.0, 0.0, step, 0);
167 }
168 size_t n = static_cast<size_t>(std::ceil(dur / step)) + 1;
169 return DoubleRange(start_, stop_, step, n);
170}
