186inline Interval& Interval::intersect(const Interval &other)
187{
188 double s = (propagate_nan::max)(start_, other.start_);
189 double t = (propagate_nan::min)(stop_, other.stop());
190 if (!(s <= t)) {
191 *this = Empty(); // 不相交 → 规范空区间（不钳回假点）
192 } else {
193 start_ = s;
194 stop_ = t;
195 }
196 return *this;
197}

199inline Interval Interval::intersected(const Interval &other) const

202 double t = (propagate_nan::min)(stop_, other.stop());
203 if (!(s <= t)) {

215inline DoubleRange Interval::discretize(double step) const
216{
217 double dur = duration();
218 if(step > 0 && dur >= 0)

220 return DoubleRange(start_, stop_, step);
