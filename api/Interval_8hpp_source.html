190inline Interval& Interval::intersect(const Interval &other)
191{
192 double s = (propagate_nan::max)(start_, other.start_);
193 double t = (propagate_nan::min)(stop_, other.stop());
194 if (!(s <= t)) {
195 *this = Empty(); // 不相交 → 规范空区间（不钳回假点）
196 } else {
197 start_ = s;
198 stop_ = t;
199 }
200 return *this;
201}

203inline Interval Interval::intersected(const Interval &other) const
204{
205 double s = (propagate_nan::max)(start_, other.start_);
206 double t = (propagate_nan::min)(stop_, other.stop());
207 if (!(s <= t)) {
208 return Empty(); // 不相交 → 规范空区间
209 }
210 return Interval{s, t};

219inline DoubleRange Interval::discretize(double step) const
